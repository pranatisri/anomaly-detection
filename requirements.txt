# Ranges, not exact pins.
#
# Exact pins (numpy==2.0.2 etc.) were tried first and broke the Streamlit Cloud deploy:
# numpy 2.0.2 ships no wheels for Python 3.13, so pip fell back to building from source
# and the installer exited non-zero at "Preparing metadata (pyproject.toml)". Ranges let
# pip pick a prebuilt wheel for whatever interpreter the platform provides.
#
# Local versions these ranges admit, and on which every reported number was produced:
# numpy 2.0.2, pandas 2.3.2, scipy 1.13.1, scikit-learn 1.5.1, lightgbm 4.6.0,
# streamlit 1.50.0, matplotlib 3.9.4, pyarrow 21.0.0.

numpy>=1.26,<3
pandas>=2.0,<3
scipy>=1.11
scikit-learn>=1.3
lightgbm>=4.0
pyarrow>=14
matplotlib>=3.7

# >=1.49 for the width="stretch" dataframe/table API used in the dashboard.
streamlit>=1.49
